-- Aggregations & Group By
SELECT MAX(salary) AS Highly_Paid_Emp FROM Worker;

SELECT count(salary) AS minSal, department 
FROM Worker 
GROUP BY department;

SELECT count(department), department 
FROM Worker 
GROUP BY department
