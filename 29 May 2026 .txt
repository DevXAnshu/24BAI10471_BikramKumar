// ---------------------------------------------------------------------------
// QUERY 1: Find products where 'price' exists, project name & price, and sort ascending.
// Note: This will return _id 7 because its price is 'null' (which technically exists).
// ---------------------------------------------------------------------------
db.products.find(
    { "price": { $exists: true } }, 
    { "name": 1, "price": 1 }
).sort(
    { "price": 1 }
);

// ---------------------------------------------------------------------------
// QUERY 2: Find products that have BOTH "black" and "white" using $and.
// This explicitly checks for both conditions being true across the array.
// ---------------------------------------------------------------------------
db.products.find(
    { 
        $and: [
            { "color": "black" },
            { "color": "white" }
        ] 
    }, 
    { "name": 1, "color": 1 }
);

// ---------------------------------------------------------------------------
// QUERY 3: Find products that have BOTH "black" and "white" using $all.
// This is the cleaner, idiomatic MongoDB way to query multi-value arrays.
// ---------------------------------------------------------------------------
db.products.find(
    { "color": { $all: ["black", "white"] } }, 
    { "name": 1, "color": 1 }
);
